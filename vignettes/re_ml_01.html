# We clean the data
flux_data_cleaned <- flux_data %>%
  # We elect only the columns we need
  select(
    TIMESTAMP,                 # Keep the timestamp
    GPP_NT_VUT_REF,            # Gross primary production
    NEE_VUT_REF_QC,            # Quality control
    TA_F,                      # Air temperature
    SW_IN_F,                   # Shortwave incoming radiation
    LW_IN_F,                   # Longwave incoming radiation
    VPD_F,                     # Vapor pressure deficit
    P_F,                       # Precipitation
    WS_F                       # Wind speed
  ) %>%
  # We rename GPP column for simplicity
  rename(GPP = GPP_NT_VUT_REF) %>%
  # We convert TIMESTAMP to date object after making it a character 
  mutate(TIMESTAMP = as.Date(as.character(TIMESTAMP), format = "%Y%m%d")) %>%
  # We replace -9999 values with NA
   mutate(across(where(is.numeric), ~na_if(.x, -9999))) %>%
  # We filter rows only to keep good quality data (value 1)
  filter(NEE_VUT_REF_QC == 1) %>%
  # We remove rows where any of the remaining columns have NA values
  select(TIMESTAMP, GPP, TA_F, SW_IN_F, LW_IN_F, VPD_F, P_F, WS_F) %>%
  filter(complete.cases(.))

3. Preprocessing

# We define the recipe and we select the numeric predictors 
gpp_recipe <- recipe(GPP ~ TA_F + SW_IN_F + LW_IN_F + VPD_F + P_F + WS_F, data = train_data) %>%
  # We apply Yeo-Johnson transformation to all numeric predictors (makes the distributions of the variables more normal-like). We use it over the Box-Cox because because Box-Cox requires strictly positive (> 0) values, and our data          includes non-positive values in TA_F (negative) and P_F (zero)
  step_YeoJohnson(all_predictors()) %>%
  # We center all numeric predictors (subtract mean)
  step_center(all_predictors()) %>%
  # We scale all numeric predictors (divide by standard deviation)
  step_scale(all_predictors())

4.1. Liner regression model evaluation on test data

# Function to calculate metrics and create plots
eval_model <- function(model, test_data, model_name) {

  # We make predictions on the test data
  predictions <- predict(model, newdata = test_data)

  # We create a data holding observed and predicted values
  eval_data <- data.frame(
    Observed = test_data$GPP,
    Predicted = predictions
  )

  # We calculate performance metrics using the built-in functions from the yardstick package
  rmse <- yardstick::rmse_vec(eval_data$Observed, eval_data$Predicted)
  rsquared <- yardstick::rsq_vec(eval_data$Observed, eval_data$Predicted)
  mae <- yardstick::mae_vec(eval_data$Observed, eval_data$Predicted)

  cat(paste0(model_name, " Performance on Test Set:\n"))
  cat(paste0("RMSE: ", round(rmse, 3), "\n"))
  cat(paste0("R-squared: ", round(rsquared, 3), "\n"))
  cat(paste0("MAE: ", round(mae, 3), "\n\n"))

  # Create scatter plot
  p <- ggplot(eval_data, aes(x = Observed, y = Predicted)) +
    geom_point(alpha = 0.5) +
    geom_abline(intercept = 0, slope = 1, linetype = "dashed", color = "red", size=1) + 
    labs(
      title = paste(model_name, ": Observed vs. Predicted GPP (test set)"),
      subtitle = paste("RMSE:", round(rmse, 2), " R²:", round(rsquared, 2)),
      x = "Observed GPP",
      y = "Predicted GPP"
    ) +
    theme_minimal() +
    coord_fixed(ratio = 1) # This ensure axes have the same scale for 1:1 comparison

  print(p)
}

6. Results interpretation

Analyzing the evaluation between the training and the test set, we can observe some interesting results. The variance didn’t increase with the more complex KNN model compared to the simpler LM model. The R² drop was smaller for the KNN model and his value also higher: - LM: CV R² ≈ 0.698 –> Test R² ≈ 0.688 (Drop ≈ 0.010) - KNN (k=8): CV R² ≈ 0.729 –> Test R² ≈ 0.727 (Drop ≈ 0.002) This means that the KNN model with k=8, despite being more flexible than the LM model,generalized well from the training data to the test data without significant overfitting.

The KNN model also had clearly better test set metrics (RMSE: 1.438 vs 1.537; R²: 0.727 vs 0.688). This means that it has a lower bias compared to the LM model. This also strongly suggests that the underlying relationship between the environmental predictors (temperature, radiation, VPD, etc.) and GPP is likely not strictly linear, so the the assumption of linearity of the LM model doesn’t fully capture the real-world complexity unlike the more flexible KNN model.

So in brief, the linear regression model is positioned more towards the high bias low variance end of the spectrum because of its rigid linear assumption preventing it from fitting the data perfectly. This inherent simplicity also means it’s less sensitive to noise in the training data, resulting in low variance, as evidenced by the small drop in performance from CV to test. The K-Nearest Neighbors model positioned more towards the lower bias end of the spectrum by being flexible enough to capture more of the true relationship. While more flexible models like KNN can be prone to higher variance, for this specific dataset and the choice of k=8 appears to have effectively managed this variance, leading to minimal overfitting, as shown by the very small drop in performance between CV and test results

plot_data_for_timeseries <- time_series_plot_data %>%
  select(
    Date = TIMESTAMP,          
    Observed_GPP = GPP,        
    LM_Predicted_GPP = lm_predictions,
    KNN_Predicted_GPP = knn_predictions
  ) %>%
  arrange(Date) # Ensure data is sorted by Date 
# We reshape the data from wide to long format for ggplot2. Now we have a column for dates, a column "Source" which tells where each "GPP_Value" came from (observed, LM or KNN) and a "GPP_Value" column
plot_data_long <- plot_data_for_timeseries %>%
  pivot_longer(
    cols = c("Observed_GPP", "LM_Predicted_GPP", "KNN_Predicted_GPP"),                   # List the columns that we want to stack up
    names_to = "Source", 
    values_to = "GPP_Value" 
  )
